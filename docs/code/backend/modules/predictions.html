 56        coverage_ratio = 0.0
 57
 58        window_start = current_time_t - timedelta(
 59            minutes=self.settings.app.ml_telemetry_window_minutes
 60        )
 61        window_end = current_time_t
 62
 63        if history:
 64            speeds = [p.speed for p in history]
 65            segment_avg_speed = sum(speeds) / len(speeds)
 66
 67            history_sorted = sorted(history, key=lambda x: x.timestamp)
 68
 69            for i in range(1, len(history_sorted)):
 70                delta = (
 71                    history_sorted[i].timestamp - history_sorted[i - 1].timestamp
 72                ).total_seconds()
 73
 74                if delta > self.settings.app.max_idle_gap_s:
 75                    continue
 76
 77                if history_sorted[i].speed < self.settings.app.stop_speed_threshold_kmh:
 78                    idle_time_s += max(0, delta)
 79
 80            window_points = [
 81                p for p in history_sorted if window_start <= p.timestamp <= window_end
 82            ]
 83
 84            telemetry_points = [
 85                TelemetryPoint(
 86                    timestamp=p.timestamp,
 87                    longitude=p.longitude,
 88                    latitude=p.latitude,
 89                    speed=float(p.speed),
 90                    course=float(p.course),
 91                    location_valid=p.location_valid,
 92                    packet_time=p.packet_time if p.packet_time else p.timestamp,
 93                    is_historical=p.is_historical,
 94                )
 95                for p in window_points
 96            ]
 97
 98            expected_points = (
 99                self.settings.app.ml_telemetry_window_minutes * 60
100            ) / 10.0
101            coverage_ratio = min(1.0, len(window_points) / expected_points)
102
103        cur_dev_s = 0.0
104        previous_stop = await self.schedule.get_previous_stop(tr_id, current_time_t)
105
106        if (
107            previous_stop
108            and previous_stop.longitude
109            and previous_stop.latitude
110            and history_sorted
111        ):
112            for p in reversed(history_sorted):
113                dist = geopy_distance(
114                    (p.latitude, p.longitude),
115                    (previous_stop.latitude, previous_stop.longitude),
116                ).meters
117
118                if dist < 50.0 and p.speed < self.settings.app.stop_speed_threshold_kmh:
119                    cur_dev_s = (p.timestamp - previous_stop.time_begin).total_seconds()
120                    break
121
122        next_stop = await self.schedule.get_next_stop(tr_id, current_time_t)
123        if not next_stop:
124            raise ValueError(f"No upcoming stops in schedule for tr_id {tr_id}")
125
126        route_features = None
127        if next_stop.longitude is not None and next_stop.latitude is not None:
128            stop_times = await self.schedule.get_stop_times(tr_id)
129            now = current_time_t.timestamp()
130            target_time = next_stop.time_begin.timestamp()
131
132            valid_points = [
133                p
134                for p in window_points
135                if p.location_valid
136                and all(
137                    math.isfinite(v)
138                    for v in (p.longitude, p.latitude, float(p.speed), float(p.course))
139                )
140                and -180 <= p.longitude <= 180
141                and -90 <= p.latitude <= 90
142                and p.longitude != 0
143                and p.latitude != 0
144                and 0 <= float(p.speed) <= 130
145            ]
146
147            if valid_points:
148                latest = valid_points[-1]
149                lon = latest.longitude
150                lat = latest.latitude
151                speed = float(latest.speed)
152                heading = float(latest.course)
153                gps_age_s = now - latest.timestamp.timestamp()
154
155                lat1, lat2 = math.radians(lat), math.radians(next_stop.latitude)
156                h = (
157                    math.sin((lat2 - lat1) / 2) ** 2
158                    + math.cos(lat1)
159                    * math.cos(lat2)
160                    * math.sin(math.radians(next_stop.longitude - lon) / 2) ** 2
161                )
162                distance_target_m = (
163                    6371000 * 2 * math.asin(min(1, math.sqrt(max(0, h))))
164                )
165
166                east = (
167                    math.radians(next_stop.longitude - lon)
168                    * 6371000
169                    * math.cos(math.radians((lat + next_stop.latitude) / 2))
170                )
171                north = math.radians(next_stop.latitude - lat) * 6371000
172                bearing = math.degrees(math.atan2(east, north))
173
174                heading_alignment = math.cos(math.radians(bearing - heading))
175            else:
176                lon = math.nan
177                lat = math.nan
178                speed = math.nan
179                heading_alignment = math.nan
180                distance_target_m = math.nan
181                east = math.nan
182                north = math.nan
183                gps_age_s = 3600.0
184
185            route_features = RouteFeatures(
186                cur_dev_missing=0.0,
187                horizon_s=target_time - now,
188                hour_sin=math.sin(2 * math.pi * (now % 86400) / 86400),
189                hour_cos=math.cos(2 * math.pi * (now % 86400) / 86400),
190                stops_ahead=float(
191                    bisect_right(stop_times, target_time)
192                    - bisect_right(stop_times, now)
193                ),
194                target_lon=next_stop.longitude,
195                target_lat=next_stop.latitude,
196                distance_target_m=distance_target_m,
197                gps_age_s=gps_age_s,
198                last_lon=lon,
199                last_lat=lat,
200                last_speed=speed,
201                heading_alignment=heading_alignment,
202                target_east_m=east,
203                target_north_m=north,
204            )
205
206        ml_request = MLPredictionRequest(
207            tr_id=tr_id,
208            target_stop_id=next_stop.stop_id,
209            target_time_begin=next_stop.time_begin,
210            current_time_T=current_time_t,
211            cur_dev_s=cur_dev_s,
212            aggregates=TelemetryAggregates(
213                segment_avg_speed=segment_avg_speed,
214                idle_time_s=idle_time_s,
215                coverage_ratio=coverage_ratio,
216            ),
217            window=TelemetryWindow(
218                start_time=window_start,
219                end_time=window_end,
220                recent_points=telemetry_points,
221            ),
222            route_features=route_features,
223        )
224
225        ml_resp, error_text = await self.ml_service.get_prediction(
226            ml_request, fallback_delay_s=cur_dev_s
227        )
228
229        log = await self.repository.save_prediction(
230            tr_id=tr_id,
231            predicted_delay_s=ml_resp.predicted_delay_s,
232            risk_level=ml_resp.risk_level,
233            pattern_reason=ml_resp.pattern_reason,
234            error_text=error_text,
235            status=ml_resp.status,
236        )
237
238        dto = PredictionLogDto(
239            id=log.id,
240            tr_id=log.tr_id,
241            predicted_delay_s=log.predicted_delay_s,
242            risk_level=RiskLevel(log.risk_level),
243            pattern_reason=IncidentPattern(log.pattern_reason)
244            if log.pattern_reason
245            else None,
246            created_at=log.created_at,
247        )
248
249        await self.redis.set(
250            f"vehicle_prediction:{tr_id}",
251            dto.model_dump_json(),
252            ex=self.settings.app.prediction_cache_ttl,
253        )
254
255        return dto
